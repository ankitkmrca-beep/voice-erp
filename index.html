async function extractFieldsWithGemini(transcriptText, geminiApiKey) { const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${geminiApiKey}`; const prompt = `You are a sales entry assistant for an Indian ERP billing application. Extract sale details from this spoken text: "${transcriptText}". Rules: 1. "Ek AC" / "One AC" / "AC" means qty: 1 and itemDesc: "AC". 2. If customer name is missing, use "Cash Customer". 3. If rate or quantity is missing, return 0 or 1 respectively. 4. Return ONLY valid JSON with these exact keys: { "customerName": string, "itemDesc": string, "qty": number, "rate": number, "paymentStatus": "Paid" or "Pending" }`; try { const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { responseMimeType: "application/json" } }) }); const data = await response.json(); const rawJsonText = data.candidates[0].content.parts[0].text; const parsedData = JSON.parse(rawJsonText); // Populate your ERP form fields fillFormFields( parsedData.customerName || "Cash Customer", parsedData.itemDesc || "General Goods", parsedData.qty || 1, parsedData.rate || 0, parsedData.paymentStatus || "Paid" ); } catch (error) { console.error("Gemini API Error:", error); } }
