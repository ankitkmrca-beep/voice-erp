let recognition = null; let isRecording = false; function startVoiceEntry() { const statusText = document.getElementById("statusText"); const micBtn = document.querySelector(".mic-btn"); // Check if browser supports Speech Recognition const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { alert("Speech recognition is not supported in this browser. Please use Chrome."); return; } // If already recording, TAP AGAIN TO STOP if (isRecording && recognition) { recognition.stop(); return; } // Initialize Speech Recognition recognition = new SpeechRecognition(); recognition.continuous = true; // Keeps listening until tapped again recognition.interimResults = true; // Shows live preview as you speak recognition.lang = 'hi-IN'; // Optimized for Hindi / Hinglish let finalTranscript = ''; recognition.onstart = function() { isRecording = true; if (micBtn) micBtn.innerText = "🛑 Tap Again to Stop Recording"; statusText.innerText = "🎙️ Recording... Speak now and tap again when finished!"; }; recognition.onresult = function(event) { let interimTranscript = ''; for (let i = event.resultIndex; i < event.results.length; ++i) { if (event.results[i].isFinal) { finalTranscript += event.results[i][0].transcript; } else { interimTranscript += event.results[i][0].transcript; } } statusText.innerText = "Listening: " + (finalTranscript || interimTranscript); }; recognition.onerror = function(event) { statusText.innerText = "Error: " + event.error; resetMicUI(); }; recognition.onend = function() { resetMicUI(); if (finalTranscript.trim().length > 0) { statusText.innerText = `Processing: "${finalTranscript}"`; parseVoiceCommand(finalTranscript); } else { statusText.innerText = "No voice detected. Tap to try again."; } }; recognition.start(); } function resetMicUI() { isRecording = false; const micBtn = document.querySelector(".mic-btn"); if (micBtn) micBtn.innerText = "🎙️ Speak Command (Jarvis)"; }
